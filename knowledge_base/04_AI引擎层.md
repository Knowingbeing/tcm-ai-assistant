class TCMDiagnosisEngine:
    def __init__(self, api_key: str, provider: str = "DeepSeek",
                 model: str = ""):
        """
        初始化 AI 诊断引擎。
        
        参数：
        - api_key: LLM API 密钥
        - provider: 服务商名称（API_PROVIDERS 的 key）
        - model: 模型名称，为空时使用 provider 的 default_model
        """
        self.api_key = api_key.strip() if api_key else ""
        self.provider = provider
        self.model = model or API_PROVIDERS.get(
            provider, {}
        ).get("default_model", "")
        
        # 判断 API Key 是否可用
        self.has_api_key = bool(self.api_key)
        
        # 初始化 OpenAI 客户端
        self.client = None
        if self.has_api_key:
            base_url = API_PROVIDERS.get(provider, {}).get("base_url")
            try:
                self.client = OpenAI(
                    api_key=self.api_key,
                    base_url=base_url
                )
            except Exception as e:
                print(f"[LLM] 客户端初始化失败: {e}")
                self.has_api_key = False
        
        # 加载中医知识库（注入 prompt）
        self.knowledge_base = self._load_knowledge_base()
```

### 3.2 核心辨证函数 `analyze_symptoms`

```python
def analyze_symptoms(
    self,
    chief_complaint: str,
    symptoms: list[str],
    tongue_sign: str,
    pulse_sign: str
) -> dict:
    """
    核心辨证函数。根据主诉、症状、舌象、脉象进行中医辨证。
    
    返回 dict 结构：
    {
        "syndrome": "证型名称",
        "syndrome_category": "辨证体系（六经/脏腑/气血津液等）",
        "analysis": "辨证分析文本",
        "formula": "推荐方剂",
        "formula_adjustment": "方剂加减建议",
        "treatment_principle": "治法",
        "confidence": 85  # 0-100 整数
    }
    
    调用流程：
    1. 有 API Key → 构造 system prompt → 调用 LLM → 解析 JSON
    2. 无 API Key 或调用失败 → _rule_based_diagnosis() 规则兜底
    """
    if not self.has_api_key:
        return self._rule_based_diagnosis(
            chief_complaint, symptoms, tongue_sign, pulse_sign
        )
    
    # 构造 system prompt
    system_prompt = f"""你是一位经验丰富的中医师，擅长辨证论治。
请根据患者的主诉、症状、舌象、脉象进行辨证分析。

{self.knowledge_base}

请严格按照以下 JSON 格式返回结果（不要包含 markdown 代码块标记）：
{{
    "syndrome": "证型名称",
    "syndrome_category": "辨证体系",
    "analysis": "详细的辨证分析，至少100字",
    "formula": "推荐方剂名称",
    "formula_adjustment": "根据患者具体情况的加减建议",
    "treatment_principle": "治法原则",
    "confidence": 0-100的整数，表示辨证置信度
}}"""
    
    # 构造 user prompt
    user_prompt = self._build_user_prompt(
        chief_complaint, symptoms, tongue_sign, pulse_sign
    )
    
    try:
        resp = self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": system_prompt},
                {"role": "user", "content": user_prompt}
            ],
            temperature=0.3,  # 低温度保证稳定性
            max_tokens=2000,
        )
        
        content = resp.choices[0].message.content.strip()
        # 清理可能的 markdown 代码块标记
        if content.startswith("```"):
            content = content.split("\n", 1)[-1].rsplit("```", 1)[0].strip()
        
        result = json.loads(content)
        
        # 字段校验与默认值
        result.setdefault("syndrome", "待辨证")
        result.setdefault("syndrome_category", "待分类")
        result.setdefault("analysis", "")
        result.setdefault("formula", "待推荐")
        result.setdefault("formula_adjustment", "")
        result.setdefault("treatment_principle", "")
        result.setdefault("confidence", 60)
        
        # confidence 强制为 int
        try:
            result["confidence"] = int(result["confidence"])
        except (ValueError, TypeError):
            result["confidence"] = 60
        
        return result
        
    except json.JSONDecodeError:
        # LLM 返回非 JSON → 规则兜底
        return self._rule_based_diagnosis(
            chief_complaint, symptoms, tongue_sign, pulse_sign
        )
    except Exception as e:
        err = str(e).lower()
        # 根据错误类型返回提示
        if "401" in err or "invalid api key" in err:
            return self._error_result("API Key 无效，请检查设置")
        if "404" in err or "model" in err:
            return self._error_result(f"模型 {self.model} 不可用")
        if "429" in err or "rate limit" in err:
            return self._error_result("请求频率超限，请稍后重试")
        # 网络错误 → 规则兜底
        return self._rule_based_diagnosis(
            chief_complaint, symptoms, tongue_sign, pulse_sign
        )
```

### 3.3 规则兜底 `_rule_based_diagnosis`

```python
def _rule_based_diagnosis(
    self,
    chief_complaint: str,
    symptoms: list[str],
    tongue_sign: str,
    pulse_sign: str
) -> dict:
    """
    规则引擎兜底辨证。当 API Key 未配置或 LLM 调用失败时使用。
    
    通过关键词匹配 20+ 证型，返回辨证结果。
    匹配规则：症状关键词 + 舌象 + 脉象 联合判定。
    """
    text = f"{chief_complaint} {' '.join(symptoms)} {tongue_sign} {pulse_sign}"
    
    # 证型匹配规则（部分示例）
    rules = [
        {
            "name": "太阳伤寒证",
            "category": "六经辨证",
            "keywords": ["恶寒", "发热", "无汗", "头痛", "身痛"],
            "tongue": "苔白",
            "pulse": "浮紧",
            "formula": "麻黄汤",
            "treatment": "发汗解表、宣肺平喘",
            "confidence": 75
        },
        {
            "name": "太阳中风证",
            "category": "六经辨证",
            "keywords": ["发热", "汗出", "恶风", "头痛"],
            "tongue": "苔白",
            "pulse": "浮缓",
            "formula": "桂枝汤",
            "treatment": "解肌发表、调和营卫",
            "confidence": 75
        },
        # ... 共 20+ 条规则
        {
            "name": "肝气郁结证",
            "category": "脏腑辨证",
            "keywords": ["胸胁胀痛", "善太息", "情志抑郁", "烦躁"],
            "tongue": "苔白",
            "pulse": "弦",
            "formula": "逍遥散",
            "treatment": "疏肝解郁",
            "confidence": 70
        },
        {
            "name": "肾阴虚证",
            "category": "脏腑辨证",
            "keywords": ["腰膝酸软", "头晕", "耳鸣", "潮热", "盗汗"],
            "tongue": "舌红少苔",
            "pulse": "细数",
            "formula": "六味地黄丸",
            "treatment": "滋补肾阴",
            "confidence": 70
        },
        # ...
    ]
    
    # 匹配算法
    best_match = None
    best_score = 0
    
    for rule in rules:
        score = 0
        # 症状关键词匹配（每个匹配 +2 分）
        for kw in rule["keywords"]:
            if kw in text:
                score += 2
        # 舌象匹配（+3 分）
        if rule["tongue"] and rule["tongue"] in tongue_sign:
            score += 3
        # 脉象匹配（+3 分）
        if rule["pulse"] and rule["pulse"] in pulse_sign:
            score += 3
        
        if score > best_score:
            best_score = score
            best_match = rule
    
    if best_match and best_score >= 3:
        return {
            "syndrome": best_match["name"],
            "syndrome_category": best_match["category"],
            "analysis": f"基于规则引擎匹配：{best_match['name']}。"
                       f"匹配关键词得分：{best_score}。",
            "formula": best_match["formula"],
            "formula_adjustment": "",
            "treatment_principle": best_match["treatment"],
            "confidence": best_match["confidence"]
        }
    
    # 无匹配
    return {
        "syndrome": "待辨证",
        "syndrome_category": "待分类",
        "analysis": "症状信息不足，无法通过规则引擎匹配证型。"
                   "请配置 API Key 以使用 AI 辨证。",
        "formula": "待推荐",
        "formula_adjustment": "",
        "treatment_principle": "",
        "confidence": 0
    }
```

### 3.4 多轮追问 `should_ask_followup`

```python
def should_ask_followup(
    self,
    chief_complaint: str,
    symptoms: list[str],
    tongue_sign: str,
    pulse_sign: str,
    round_count: int = 0
) -> dict:
    """
    判断是否需要追问，并生成追问问题。
    
    追问维度（5 个）：
    1. 舌象 (tongue_sign)
    2. 脉象 (pulse_sign)
    3. 寒热 (cold/heat)
    4. 汗出 (sweat)
    5. 二便 (bowel/urine)
    
    规则：
    - 最多追问 2 轮（round_count >= 2 时强制不再追问）
    - 每轮最多 2 个问题
    - 已有信息的维度不追问
    
    返回 dict:
    {
        "need_followup": True/False,
        "questions": [
            {
                "field": "tongue_sign",      # 写入 session_state 的 key
                "label": "请选择您的舌象",     # UI 显示
                "options": ["舌淡苔白", ...]   # 选项列表
            }
        ]
    }
    """
    # 强制不再追问
    if round_count >= 2:
        return {"need_followup": False, "questions": []}
    
    questions = []
    text = f"{chief_complaint} {' '.join(symptoms)}"
    
    # 检查舌象
    if not tongue_sign.strip():
        questions.append({
            "field": "tongue_sign",
            "label": "请选择您的舌象",
            "options": [
                "舌淡苔白", "舌红苔黄", "舌红少苔",
                "舌紫暗有瘀斑", "舌淡胖苔白", "舌红苔黄腻",
                "舌苔白腻", "舌苔薄白"
            ]
        })
    
    # 检查脉象
    if not pulse_sign.strip():
        questions.append({
            "field": "pulse_sign",
            "label": "请选择您的脉象",
            "options": [
                "脉浮紧", "脉浮缓", "脉浮数", "脉沉迟",
                "脉沉实", "脉弦", "脉弦数", "脉细数",
                "脉洪大", "脉滑", "脉涩", "脉微细"
            ]
        })
    
    # 检查寒热
    has_cold_heat = any(kw in text for kw in
        ["恶寒", "发热", "畏寒", "寒热往来", "身热"])
    if not has_cold_heat:
        questions.append({
            "field": "ten_asks_data",
            "label": "您是否有寒热症状？",
            "options": ["无寒热", "恶寒发热", "但热不寒",
                       "但寒不热", "寒热往来"]
        })
    
    # 检查汗出
    has_sweat = any(kw in text for kw in
        ["汗", "自汗", "盗汗", "无汗"])
    if not has_sweat:
        questions.append({
            "field": "ten_asks_data",
            "label": "您的出汗情况？",
            "options": ["无异常", "自汗", "盗汗", "无汗", "大汗"]
        })
    
    # 检查二便
    has_bowel = any(kw in text for kw in
        ["大便", "小便", "便秘", "腹泻", "便溏"])
    if not has_bowel:
        questions.append({
            "field": "ten_asks_data",
            "label": "您的二便情况？",
            "options": ["正常", "大便干结", "大便稀溏",
                       "小便短赤", "小便清长"]
        })
    
    # 每轮最多 2 个问题
    questions = questions[:2]
    
    return {
        "need_followup": len(questions) > 0,
        "questions": questions
    }
```

### 3.6 自由对话 `chat_with_history`

```python
def chat_with_history(
    self,
    messages: list[dict],
    temperature: float = 0.7,
    max_tokens: int = 1000
) -> str:
    """
    多轮自由对话（不限定 JSON 输出格式）。
    用于用户与 AI 的自由问答交流。
    
    参数：
    - messages: OpenAI 格式的对话历史
    - temperature: 0.7（比辨证的 0.3 高，更灵活）
    - max_tokens: 1000
    
    返回：AI 回复文本
    """
    if not self.has_api_key:
        return "未配置 API Key，无法使用对话功能。"
    
    system_msg = {
        "role": "system",
        "content": f"你是一位中医师，请用专业且通俗的语言回答问题。\n"
                   f"{self.knowledge_base}"
    }
    
    try:
        resp = self.client.chat.completions.create(
            model=self.model,
            messages=[system_msg] + messages,
            temperature=temperature,
            max_tokens=max_tokens,
        )
        return resp.choices[0].message.content.strip()
    except Exception as e:
        return f"对话失败：{str(e)}"
```

## 4. 知识库注入 `_load_knowledge_base`

### 4.1 当前实现（硬编码字符串）

def _load_knowledge_base(self) -> str:
    """

### 4.2 当前方案的局限

1. **静态注入**：每次 LLM 调用都携带完整的知识库字符串，不论是否相关
2. **内容有限**：硬编码字符串只包含证型大纲，不包含具体方剂组成、中药性味归经
3. **Token 浪费**：完整知识库约 500 token，每次调用都消耗
4. **无法检索**：不能根据患者症状动态检索相关知识，只能依赖 LLM 自身理解

### 4.3 优化方向（详见 `08_RAG策略评估与优化方案.md`）

将 `data/tcm_data.py` 中的 FORMULAS（68 首）、SYNDROMES（47 个）、HERBS（87 味）作为知识源，实现真正的 RAG 检索增强生成。

## 5. 错误处理策略

| 错误类型 | HTTP 状态码 | 处理方式 | 用户提示 |
|----------|-------------|----------|----------|
| API Key 无效 | 401 | 降级规则引擎 | "API Key 无效，请检查设置" |
| 模型不存在 | 404 | 降级规则引擎 | "模型 {model} 不可用" |
| 频率限制 | 429 | 降级规则引擎 | "请求频率超限，请稍后重试" |
| JSON 解析失败 | - | 降级规则引擎 | （静默降级，用户无感知） |
| 网络错误 | - | 降级规则引擎 | （静默降级，用户无感知） |
| 未知错误 | - | 降级规则引擎 | （静默降级，用户无感知） |

**设计原则**：任何 LLM 调用失败都不应阻断用户流程，自动降级到规则引擎保证基本可用性。

def _build_user_prompt(self, chief, symptoms, tongue, pulse) -> str:
    return f"""请对以下患者进行辨证分析：

主诉：{chief}
症状：{', '.join(symptoms) if symptoms else '无'}
舌象：{tongue or '未提供'}
脉象：{pulse or '未提供'}

请给出证型、辨证体系、分析、推荐方剂、治法和置信度。"""
```
